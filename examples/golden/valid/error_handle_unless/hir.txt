  Proc f(x: S64) -> S64<>Some_Error
    Block
      If condition=Binary(< Ident(x) Int(0))
        Block
          Return StructInit(value, value=Int(0), error=Error(GENERIC), hasError=Bool(true))
      Return StructInit(value, value=Binary(* Ident(x) Int(2)), error=Error(0), hasError=Bool(false))
  Proc main() -> S64
    Block
      Block
        VarDecl : S64<>Some_Error = Call(f, Int(-1)) [mutable=false compileTime=false]
        IfCatch Ident(__chaos_union) catch err
          Block
            If condition=Binary(== Ident(err) Error(GENERIC))
              Block
                Return Int(7)
            Return Int(8)
        VarDecl r: S64 = FieldLoad(Ident(__chaos_union), 0) [mutable=true compileTime=false]
      Return Ident(r)
