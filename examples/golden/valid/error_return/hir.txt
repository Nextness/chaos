HIR
  Proc double(x: S64) -> S64<>Some_Error
    Block
      If condition=Binary(< Ident(x) Int(0))
        Block
          Return StructInit(value, value=Unknown, error=Error(GENERIC), hasError=Bool(true))
      Return StructInit(value, value=Binary(* Ident(x) Int(2)), error=Unknown, hasError=Bool(false))
  Proc triple() -> S64<>Some_Error
    Block
      Return StructInit(value, value=Int(3), error=Unknown, hasError=Bool(false))
  Proc main() -> S64
    Block
      Block
        VarDecl a: S64<>Some_Error = Call(double, Int(5)) [mutable=true compileTime=false]
        IfCatch Ident(a)
          Block
            Return Unary(- Int(1))
      Block
        VarDecl b: S64<>Some_Error = Call(triple) [mutable=true compileTime=false]
        IfCatch Ident(b)
          Block
            Return Unary(- Int(2))
      If condition=Binary(&& Binary(== FieldLoad(Ident(a), 0) Int(10)) Binary(== FieldLoad(Ident(b), 0) Int(3)))
        Block
          Return Int(10)
      Return Int(0)
