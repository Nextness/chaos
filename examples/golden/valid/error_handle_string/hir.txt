HIR
  Proc greet(x: S64) -> String<>Some_Error
    Block
      If condition=Binary(< Ident(x) Int(0))
        Block
          Return StructInit(value, value=String(""), error=Error(GENERIC), hasError=Bool(true))
      Return StructInit(value, value=String("hello"), error=Error(0), hasError=Bool(false))
  Proc main() -> S64
    Block
      Block
        VarDecl : String<>Some_Error = Call(greet, Int(1)) [mutable=false compileTime=false]
        IfCatch Ident(__chaos_union)
          Block
            Return Int(-1)
        VarDecl s: String = FieldLoad(Ident(__chaos_union), 0) [mutable=true compileTime=false]
      If condition=Binary(== Ident(s) String("hello"))
        Block
          Return Int(11)
      Return Int(0)
