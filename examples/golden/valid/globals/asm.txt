g_A:
    rb 8
g_B:

segment readable executable

entry _start

    call f_main
    mov rdi, rax
    mov rax, 60

f_main:

    mov rbp, rsp
    sub rsp, 16
L0:

    mov rax, qword [rbp-8]
    leave
    ret

f___global_init:

    mov rbp, rsp
    sub rsp, 32
L1:

    mov rax, 4
